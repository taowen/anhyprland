cmake_minimum_required(VERSION 3.22)
project(lua VERSION 5.5.0 LANGUAGES C)
include(GNUInstallDirs)
file(GLOB sources "${LUA_SOURCE_DIR}/*.c")
list(FILTER sources EXCLUDE REGEX "/(lua|luac|onelua|ltests)\\.c$")
add_library(lua SHARED ${sources})
target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
target_include_directories(lua PUBLIC ${LUA_SOURCE_DIR})
target_link_libraries(lua PRIVATE dl m)
set_target_properties(lua PROPERTIES OUTPUT_NAME lua5.5)
install(TARGETS lua)
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/lua.hpp ${LUA_SOURCE_DIR}/lua.h ${LUA_SOURCE_DIR}/luaconf.h ${LUA_SOURCE_DIR}/lualib.h ${LUA_SOURCE_DIR}/lauxlib.h DESTINATION include/lua5.5)
configure_file(lua5.5.pc.in lua5.5.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/lua5.5.pc DESTINATION lib/pkgconfig)
